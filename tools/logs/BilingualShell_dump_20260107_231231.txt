const UNCOMPRESSED_TS = new Set<string>([
  "1.2.840.10008.1.2",     // Implicit VR Little Endian
  "1.2.840.10008.1.2.1",   // Explicit VR Little Endian
  "1.2.840.10008.1.2.2"    // Explicit VR Big Endian
]);

function clamp(n: number, lo: number, hi: number) {
  return Math.max(lo, Math.min(hi, n));
}

function pickFirstNumber(s: string | undefined | null): number | null {
  if (!s) return null;
  const first = s.split("\\")[0]?.trim();
  if (!first) return null;
  const n = Number(first);
  return Number.isFinite(n) ? n : null;
}

async function readHeader(file: File): Promise<SliceHeader> {
  const ab = await file.arrayBuffer();
  const bytes = new Uint8Array(ab);
  const ds = dicomParser.parseDicom(bytes);

  const rows = ds.uint16("x00280010") || 0;
  const cols = ds.uint16("x00280011") || 0;

  const bitsAllocated = ds.uint16("x00280100") || 16;
  const pixelRepresentation = ds.uint16("x00280103") || 0;

  const slope = Number(ds.string("x00281053") || "1") || 1;
  const intercept = Number(ds.string("x00281052") || "0") || 0;

  const photometric = (ds.string("x00280004") || "MONOCHROME2").toUpperCase();
  const transferSyntax = ds.string("x00020010") || "";

  const instance = Number(ds.string("x00200013") || ds.string("x00200011") || "0") || 0;

  const wc = pickFirstNumber(ds.string("x00281050"));
  const ww = pickFirstNumber(ds.string("x00281051"));

  const compressed = !UNCOMPRESSED_TS.has(transferSyntax);

  return {
    file,
    instance,
    rows,
    cols,
    bitsAllocated,
    pixelRepresentation,
    slope,
    intercept,
    photometric,
    transferSyntax,
    windowCenter: wc,
    windowWidth: ww,
    compressed
  };
}

function bytesToWordArray(
  pixelBytes: Uint8Array,
  signed: boolean,
  bigEndian: boolean
): Int16Array | Uint16Array {
  let buf: ArrayBuffer;
  let byteOffset = pixelBytes.byteOffset;

  if (byteOffset % 2 !== 0) {
    const copy = new Uint8Array(pixelBytes.length);
    copy.set(pixelBytes);
    buf = copy.buffer;
    byteOffset = 0;
    pixelBytes = copy;
  } else {
    buf = pixelBytes.buffer;
  }

  if (!bigEndian) {
    return signed
      ? new Int16Array(buf, byteOffset, Math.floor(pixelBytes.length / 2))
      : new Uint16Array(buf, byteOffset, Math.floor(pixelBytes.length / 2));
  }

  const n = Math.floor(pixelBytes.length / 2);
  const out = new Uint8Array(n * 2);
  for (let i = 0; i < n; i++) {
    out[i * 2] = pixelBytes[i * 2 + 1];
    out[i * 2 + 1] = pixelBytes[i * 2];
  }
  return signed ? new Int16Array(out.buffer) : new Uint16Array(out.buffer);
}

function windowToByte(v: number, wc: number, ww: number): number {
  const lo = wc - ww / 2;
  const hi = wc + ww / 2;
  if (hi <= lo) return 0;
  const x = (v - lo) / (hi - lo);
  const b = Math.round(x * 255);
  return b < 0 ? 0 : b > 255 ? 255 : b;
}

async function decodeToImageData(
  hdr: SliceHeader,
  wc: number | null,
  ww: number | null
): Promise<{ img: ImageData; min: number; max: number; wc: number; ww: number }> {
  if (hdr.compressed) {
    throw new Error("Compressed DICOM is not supported in the mini-decoder yet (need JPEG/RLE decoder).");
  }

  const ab = await hdr.file.arrayBuffer();
  const bytes = new Uint8Array(ab);
  const ds = dicomParser.parseDicom(bytes);
  const el = (ds.elements as any).x7fe00010;
  if (!el || typeof el.dataOffset !== "number" || typeof el.length !== "number") {
    throw new Error("Missing PixelData (7FE0,0010).");
  }

  const pixelBytes = new Uint8Array(ds.byteArray.buffer, el.dataOffset, el.length);
  const bigEndian = hdr.transferSyntax === "1.2.840.10008.1.2.2";
  const signed = hdr.pixelRepresentation === 1;

  let min = Number.POSITIVE_INFINITY;
  let max = Number.NEGATIVE_INFINITY;

  const img = new ImageData(hdr.cols, hdr.rows);
  const out = img.data;

  if (hdr.bitsAllocated === 8) {
    for (let i = 0; i < hdr.rows * hdr.cols; i++) {
      const raw = pixelBytes[i];
      const hu = raw * hdr.slope + hdr.intercept;
      if (hu < min) min = hu;
      if (hu > max) max = hu;
    }
  } else {
    const arr = bytesToWordArray(pixelBytes, signed, bigEndian);
    for (let i = 0; i < arr.length; i++) {
      const raw = (arr as any)[i] as number;
      const hu = raw * hdr.slope + hdr.intercept;
      if (hu < min) min = hu;
      if (hu > max) max = hu;
    }
  }

  const useWc = (wc != null) ? wc : (min + max) / 2;
  const useWw = (ww != null) ? ww : Math.max(1, (max - min));

  const invert = hdr.photometric === "MONOCHROME1";

  if (hdr.bitsAllocated === 8) {
    for (let i = 0; i < hdr.rows * hdr.cols; i++) {
      const raw = pixelBytes[i];
      const hu = raw * hdr.slope + hdr.intercept;
      let b = windowToByte(hu, useWc, useWw);
      if (invert) b = 255 - b;
      const o = i * 4;
      out[o + 0] = b;
      out[o + 1] = b;
      out[o + 2] = b;
      out[o + 3] = 255;
    }
  } else {
    const arr = bytesToWordArray(pixelBytes, signed, bigEndian);
    for (let i = 0; i < arr.length; i++) {
      const raw = (arr as any)[i] as number;
      const hu = raw * hdr.slope + hdr.intercept;
      let b = windowToByte(hu, useWc, useWw);
      if (invert) b = 255 - b;
      const o = i * 4;
      out[o + 0] = b;
      out[o + 1] = b;
      out[o + 2] = b;
      out[o + 3] = 255;
    }
  }

  return { img, min, max, wc: useWc, ww: useWw };
}
